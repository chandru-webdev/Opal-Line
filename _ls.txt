108: export interface LabelGrid {
109:   columns: number

112:   /** True when even a 1×1 grid cannot fit the label inside the page margins. */

116: export function computeGrid(cfg: LabelConfig): LabelGrid {
117:   const page = PAGE_SIZES[cfg.page]
118:   const w = Math.max(5, cfg.widthMm)
119:   const h = Math.max(5, cfg.heightMm)
120:   const margin = clamp(cfg.marginMm, 0, 40)
121:   const gap = clamp(cfg.gapMm, 0, 20)
122:   const usableW = page.widthMm - margin * 2
123:   const usableH = page.heightMm - margin * 2
124:   const columns = Math.max(1, Math.floor((usableW + gap) / (w + gap)))
125:   const rows = Math.max(1, Math.floor((usableH + gap) / (h + gap)))
126:   const overflow = w + margin * 2 > page.widthMm || h + margin * 2 > page.heightMm
127:   return { columns, rows, labelsPerPage: columns * rows, overflow }
128: }

151: // bwip-js passes any option straight to the BWIPP encoder, but its bundled
152: // types only list a subset — cast so encoder-specific options type-check.
153: type BwipSvgOptions = Parameters<typeof bwipJs.toSVG>[0]
154: function encoderOptions(options: Record<string, unknown>): BwipSvgOptions {
155:   return options as unknown as BwipSvgOptions
156: }

158: export function barcodeSvg(text: string): string | null {
159:   const value = sanitizeBarcodeText(text)
160:   if (!value) return null
161:   try {
162:     const svg = bwipJs.toSVG(
163:       encoderOptions({
164:         bcid: 'code128',
165:         text: value,
166:         height: 10,
167:         // ~10 module quiet zone on each side (required for reliable scans).
168:         paddingleft: 25,
169:         paddingright: 25,
170:         includetext: false,
171:         barcolor: '000000',
172:         backgroundcolor: 'FFFFFF',
173:       }),
174:     )
175:     // Uniform horizontal stretch keeps relative bar widths intact, so the
176:     // code stays scannable while filling the full label width.
177:     return withFillViewport(svg, 'none')
178:   } catch {
179:     return null
180:   }
181: }
182: 
183: export function qrSvg(text: string): string | null {
184:   if (!text) return null
185:   try {
186:     const svg = bwipJs.toSVG(
187:       encoderOptions({
188:         bcid: 'qrcode',
189:         text,
190:         // bwipp's option is spelled `eclevel`; M keeps codes small while
191:         // tolerating typical label damage.
192:         eclevel: 'M',
193:         scale: 1,
194:         barcolor: '1A1A2E',
195:         backgroundcolor: 'FFFFFF',
196:       }),
197:     )
198:     // Quiet zone comes from CSS padding; keep the square code undistorted.
199:     return withFillViewport(svg, 'xMidYMid meet')
200:   } catch {
201:     return null
202:   }
203: }

217: function buildLabelHtml(p: LabelProduct, cfg: LabelConfig, qrCode: string | null, barCode: string | null): string {
218:   const w = cfg.widthMm
219:   const h = cfg.heightMm
220:   const pad = clamp(Math.min(h * 0.07, w * 0.035), 0.4, 1.6)
221:   const innerW = w - pad * 2
222:   const innerH = h - pad * 2
223: 
224:   const hasBarcode = !!barCode
225:   const barH = hasBarcode ? clamp(h * 0.17, 2.6, 5.5) : 0
226:   const barGap = hasBarcode ? clamp(h * 0.03, 0.3, 1) : 0
227: 
228:   const bodyH = Math.max(1.5, innerH - barH - barGap)
229:   const qrSize = qrCode ? clamp(Math.min(bodyH, innerW * 0.38), 0, 16) : 0
230:   const showQr = !!qrCode && qrSize >= 3
231:   const qrGap = showQr ? clamp(w * 0.03, 0.4, 1.4) : 0
232: 
233:   const metaLines: string[] = []
234:   if (cfg.showCompany && cfg.companyName) metaLines.push(`<div class="company">${escapeHtml(cfg.companyName)}</div>`)
235:   if (cfg.showSku && p.sku) metaLines.push(`<div class="meta sku">${escapeHtml(p.sku)}</div>`)
236:   if (cfg.showHuid && p.huid) metaLines.push(`<div class="meta huid">HUID: ${escapeHtml(p.huid)}</div>`)
237:   if (cfg.showCategory && p.category) metaLines.push(`<div class="meta cat">${escapeHtml(p.category)}</div>`)
238: 
239:   const bottom: string[] = []
240:   if (cfg.showPrice) bottom.push(`<div class="price">${formatPrice(p.price)}</div>`)
241:   if (cfg.showWeight && p.weight) bottom.push(`<div class="weight">${p.weight}g</div>`)
242: 
243:   return `
244:     <div class="lbl">
245:       <div class="body" style="height:${bodyH}mm">
246:         ${showQr ? `<div class="qr" style="width:${qrSize}mm;height:${qrSize}mm;margin-right:${qrGap}mm">${qrCode}</div>` : ''}
247:         <div class="info">
248:           ${cfg.showName && p.name ? `<div class="name">${escapeHtml(p.name)}</div>` : ''}
249:           ${metaLines.join('')}
250:           ${bottom.length ? `<div class="bottom">${bottom.join('')}</div>` : ''}
251:         </div>
252:       </div>
253:       ${hasBarcode ? `<div class="barcode" style="height:${barH}mm;margin-top:${barGap}mm">${barCode}</div>` : ''}
254:     </div>`
255: }

270: export function expandLabels(products: LabelProduct[], cfg: LabelConfig): LabelProduct[] {
271:   const copies = clamp(Math.round(cfg.copies) || 1, 1, 99)
272:   if (copies === 1) return products
273:   const out: LabelProduct[] = []
274:   for (const p of products) for (let i = 0; i < copies; i++) out.push(p)
275:   return out
276: }

278: export function buildLabelSheet(products: LabelProduct[], cfg: LabelConfig, opts: LabelSheetOptions = {}): LabelSheetResult {
279:   const mode = opts.mode ?? 'print'
280:   const grid = computeGrid(cfg)
281:   const page = PAGE_SIZES[cfg.page]
282:   const expanded = expandLabels(products, cfg)
283:   const totalLabels = expanded.length
284:   const totalPages = Math.max(1, Math.ceil(totalLabels / grid.labelsPerPage))
285:   const cap = mode === 'preview' ? Math.min(totalLabels, opts.maxLabels ?? grid.labelsPerPage * 2) : totalLabels
286:   const rendered = expanded.slice(0, cap)
287:   const renderedPages = Math.max(1, Math.ceil(rendered.length / grid.labelsPerPage))
288: 
289:   const qrCache = new Map<string, string | null>()
290:   const barCache = new Map<string, string | null>()
291: 
292:   const pages: string[] = []
293:   for (let pageIdx = 0; pageIdx < renderedPages; pageIdx++) {
294:     const slice = rendered.slice(pageIdx * grid.labelsPerPage, (pageIdx + 1) * grid.labelsPerPage)
295:     const cells: string[] = []
296:     for (const p of slice) {
297:       let qr: string | null = null
298:       if (cfg.showQr) {
299:         const key = qrPayload(p)
300:         if (!qrCache.has(key)) qrCache.set(key, qrSvg(key))
301:         qr = qrCache.get(key) ?? null
302:       }
303:       let bar: string | null = null
304:       if (cfg.showBarcode) {
305:         const value = barcodeValue(p)
306:         if (!barCache.has(value)) barCache.set(value, barcodeSvg(value))
307:         bar = barCache.get(value) ?? null
308:       }
309:       cells.push(buildLabelHtml(p, cfg, qr, bar))
310:     }
311:     pages.push(`<div class="page">${cells.join('')}</div>`)
312:   }
313: 
314:   const borderCss = cfg.showBorder ? 'border:0.15mm solid #9aa1ad;' : 'border:none;'
315: 
316:   const screenCss = `@media screen{
317:       body{background:#e8ebf0;padding:6mm;}
318:       .page{margin:0 auto 6mm;box-shadow:0 1px 6px rgba(15,23,42,.18);background:#fff;}
319:     }`
320: 
321:   const html = `<!doctype html><html><head><meta charset="utf-8"><title>Barcode Labels</title><style>
322:     @page{size:${cfg.page === 'letter' ? 'letter' : 'A4'};margin:0;}
323:     *{margin:0;padding:0;box-sizing:border-box;}
324:     html,body{background:#fff;color:#111827;-webkit-print-color-adjust:exact;print-color-adjust:exact;}
325:     body{font-family:"Segoe UI",Arial,Helvetica,sans-serif;}
326:     .page{
327:       width:${page.widthMm}mm;height:${page.heightMm}mm;padding:${cfg.marginMm}mm;
328:       display:grid;grid-template-columns:repeat(${grid.columns},${cfg.widthMm}mm);
329:       grid-auto-rows:${cfg.heightMm}mm;gap:${cfg.gapMm}mm;
330:       align-content:start;justify-content:center;
331:       page-break-after:always;break-after:page;overflow:hidden;background:#fff;
332:     }
333:     .page:last-child{page-break-after:auto;break-after:auto;}
334:     .lbl{
335:       width:${cfg.widthMm}mm;height:${cfg.heightMm}mm;${borderCss}
336:       border-radius:0.8mm;overflow:hidden;background:#fff;position:relative;
337:       display:flex;flex-direction:column;padding:${padFor(cfg)}mm;
338:     }
339:     .lbl .body{display:flex;align-items:stretch;min-height:0;}
340:     .lbl .qr{flex:none;padding:0.4mm;background:#fff;}
341:     .lbl .qr svg{display:block;width:100%;height:100%;}

345:       font-weight:700;line-height:1.15;color:#1f2937;

347:       word-break:break-word;
348:     }
349:     .lbl .meta{
350:       font-size:${clamp(cfg.heightMm * 0.075 * cfg.fontScale, 1.1, 2.6)}mm;
351:       color:#6b7280;line-height:1.2;white-space:nowrap;overflow:hidden;text-overflow:ellipsis;
352:     }
353:     .lbl .meta.huid{color:#b45309;font-weight:600;}
